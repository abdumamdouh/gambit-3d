# M1 Play — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A playable, deployed 3D chess game vs Stockfish (levels 1–8) with full rules, clocks, undo and 4 camera presets, at the look-prototype quality.

**Architecture:** One zustand store owns the only `Chess` instance and a stable piece list (ids survive moves so the scene can animate). A Web Worker runs Stockfish over UCI. R3F components render the store; they never decide legality.

**Tech Stack:** Vite 6, React 19, TS 5.8, three 0.187, @react-three/fiber 9, drei 10, @react-three/postprocessing 3, chess.js 1.4, zustand 5, stockfish 17.1 (lite single), vitest 3.

**Spec:** `docs/superpowers/specs/2026-11-23-gambit-3d-design.md`

## Global Constraints

- chess.js is the single source of truth for legality.
- Engine: `stockfish-17.1-lite-single-03e3232.{js,wasm}` served from `/engine/`.
- Levels exactly as spec §5 (UCI_Elo floor 1320).
- No backend. No paid services. GPL-3.0.
- 0 console errors on load and during play.
- Layout works at 1280×800 and 390×844.

## Interfaces (shared names)

```ts
// src/game/pieces.ts
export type Color = 'w' | 'b'
export type PieceType = 'p' | 'n' | 'b' | 'r' | 'q' | 'k'
export interface Piece { id: string; type: PieceType; color: Color; square: string; captured: boolean }
export function initialPieces(): Piece[]
export function applyMove(pieces: Piece[], m: { from: string; to: string; flags: string; promotion?: string }): Piece[]

// src/game/result.ts
export interface Result { over: boolean; reason?: 'checkmate'|'stalemate'|'threefold'|'insufficient'|'fifty'|'timeout'|'resign'; winner?: Color | null }
export function gameResult(chess: Chess): Result

// src/engine/levels.ts
export interface Level { id: number; label: string; skill?: number; elo?: number; depth?: number; movetime?: number; randomness?: number }
export const LEVELS: Level[]
export function setupCommands(l: Level): string[]
export function goCommand(l: Level): string

// src/engine/uci.ts
export function parseBestMove(line: string): { from: string; to: string; promotion?: string } | null
export function parseInfo(line: string): { depth: number; scoreCp?: number; mate?: number; pv: string[] } | null
```

import { Chess } from 'chess.js'
import { initialPieces, applyMove } from './pieces'
import { gameResult } from './result'

const play = (sans: string[]) => {
  const c = new Chess(); let p = initialPieces()
  for (const s of sans) p = applyMove(p, c.move(s))
  return { c, p }
}
const at = (p: ReturnType<typeof initialPieces>, sq: string) => p.find(x => !x.captured && x.square === sq)

describe('pieces', () => {
  it('starts with 32 pieces with unique ids', () => {
    const p = initialPieces(); expect(p).toHaveLength(32); expect(new Set(p.map(x => x.id)).size).toBe(32)
  })
  it('keeps id on a normal move', () => {
    const { p } = play(['e4']); expect(at(p, 'e4')!.id).toBe(at(initialPieces(), 'e2')!.id)
  })
  it('marks capture', () => {
    const { p } = play(['e4', 'd5', 'exd5']); expect(p.filter(x => x.captured)).toHaveLength(1)
  })
  it('moves the rook when castling short', () => {
    const { p } = play(['e4', 'e5', 'Nf3', 'Nc6', 'Bc4', 'Bc5', 'O-O']); expect(at(p, 'g1')!.type).toBe('k'); expect(at(p, 'f1')!.type).toBe('r')
  })
  it('removes the pawn on en passant', () => {
    const { p } = play(['e4', 'a6', 'e5', 'd5', 'exd6']); expect(at(p, 'd5')).toBeUndefined(); expect(at(p, 'd6')!.color).toBe('w')
  })
  it('changes type on promotion', () => {
    const c = new Chess('8/P7/8/8/8/8/8/k6K w - - 0 1')
    const p0 = [{ id: 'x', type: 'p' as const, color: 'w' as const, square: 'a7', captured: false }]
    expect(applyMove(p0, c.move('a8=Q'))[0]).toMatchObject({ square: 'a8', type: 'q' })
  })
})

describe('result', () => {
  it('detects checkmate', () => { const { c } = play(['f3', 'e5', 'g4', 'Qh4#']); expect(gameResult(c)).toEqual({ over: true, reason: 'checkmate', winner: 'b' }) })
  it('detects stalemate', () => { expect(gameResult(new Chess('7k/5Q2/6K1/8/8/8/8/8 b - - 0 1'))).toMatchObject({ over: true, reason: 'stalemate', winner: null }) })
  it('detects insufficient material', () => { expect(gameResult(new Chess('8/8/8/8/8/8/8/k6K w - - 0 1')).reason).toBe('insufficient') })
  it('detects threefold', () => { const { c } = play(['Nf3', 'Nf6', 'Ng1', 'Ng8', 'Nf3', 'Nf6', 'Ng1', 'Ng8']); expect(gameResult(c).reason).toBe('threefold') })
  it('ongoing game is not over', () => { expect(gameResult(new Chess()).over).toBe(false) })
})
